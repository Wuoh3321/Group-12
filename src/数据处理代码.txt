#图谱生成代码
import pandas as pd
import networkx as nx
import matplotlib.pyplot as plt
import os

# --- 1. 配置区域 ---
# 确保这3个文件在你的代码同级目录下
NODE_FILE = 'network_metrics_keyword.csv' 
EDGE_FILE = 'vos_output_keyword_cooccurrence.csv'
COLLAB_FILE = 'vos_output_author_collaboration.csv' # 之前你上传过这个文件

OUTPUT_DIR = 'outputs/figures'
os.makedirs(OUTPUT_DIR, exist_ok=True)

# --- 2. 数据读取与预处理 ---
print("🔄 正在读取并处理数据...")

# 读取节点数据 (获取节点大小)
try:
    df_nodes = pd.read_csv(NODE_FILE)
    # 提取前100个最核心的词 (防止图太乱)
    df_nodes_top = df_nodes.nlargest(100, 'Degree')
    node_size_dict = df_nodes_top.set_index('Node')['Degree'].to_dict()
except Exception as e:
    print(f" 读取节点文件失败: {e}")
    df_nodes_top = pd.DataFrame()
    node_size_dict = {}

# 读取边数据 (关键词共现矩阵)
try:
    df_edges = pd.read_csv(EDGE_FILE)
    # 只保留权重前 150 的连线 (保留核心结构，防止图变成毛线团)
    df_edges_top = df_edges.nlargest(150, 'Weight')
except Exception as e:
    print(f" 读取边文件失败: {e}")
    df_edges_top = pd.DataFrame()

# 读取合作网络数据
try:
    df_collab = pd.read_csv(COLLAB_FILE)
    # 只保留合作次数大于1的强关系 (过滤噪音)
    df_collab_strong = df_collab[df_collab['Weight'] > 1]
except Exception as e:
    print(f" 读取合作网络文件失败: {e}")
    df_collab_strong = pd.DataFrame()

# --- 3. 绘图函数 ---
def draw_network(G, title, filename, node_color='lightblue', node_sizes=None):
    plt.figure(figsize=(12, 10))
    
    # 使用 spring_layout 布局，k值控制间距
    pos = nx.spring_layout(G, k=0.3, iterations=30) 
    
    # 绘制节点
    nx.draw_networkx_nodes(G, pos, 
                          node_color=node_color, 
                          alpha=0.8, 
                          node_size=node_sizes if node_sizes else 100)
    
    # 绘制边
    nx.draw_networkx_edges(G, pos, alpha=0.5, edge_color='gray')
    
    # 绘制标签 (只显示核心节点的标签)
    # 策略：如果节点度数>10，或者在前50个高频词里，就显示标签
    labels = {}
    for node in G.nodes():
        # 简单策略：节点度数大于阈值或者是核心词
        if G.degree(node) > 5 or (node in node_size_dict and node_size_dict.get(node, 0) > 50):
            labels[node] = node
    
    nx.draw_networkx_labels(G, pos, labels, font_size=8, font_family='SimHei')
    
    plt.title(title, fontsize=16, fontfamily='SimHei')
    plt.axis('off')
    plt.tight_layout()
    plt.savefig(os.path.join(OUTPUT_DIR, filename), dpi=150, bbox_inches='tight')
    plt.close()
    print(f"✅ 已生成: {filename}")

# --- 4. 生成 4 张图 ---

# --- 图 1: 关键词共现网络 (真实数据) ---
if not df_edges_top.empty:
    print("1/4 正在绘制 关键词共现网络...")
    G_cooc = nx.from_pandas_edgelist(df_edges_top, 'Node 1', 'Node 2', edge_attr='Weight')
    
    # 设置节点大小 (如果节点在指标文件里，用Degree；否则用默认值)
    sizes_1 = [node_size_dict.get(node, 50) * 2 for node in G_cooc.nodes()]
    
    draw_network(G_cooc, '关键词共现网络', 'keyword_cooccurrence.png', 'skyblue', sizes_1)

# --- 图 2: 作者合作网络 (真实数据) ---
if not df_collab_strong.empty:
    print("2/4 正在绘制 作者合作网络...")
    G_collab = nx.from_pandas_edgelist(df_collab_strong, 'Node 1', 'Node 2', edge_attr='Weight')
    
    # 只保留最大的连通分量 (去掉孤立点)
    G_collab = G_collab.subgraph(sorted(nx.connected_components(G_collab), key=len, reverse=True)[0])
    
    sizes_2 = [G_collab.degree(n) * 50 for n in G_collab.nodes()] # 根据合作次数定大小
    draw_network(G_collab, '作者合作网络', 'author_collaboration.png', 'lightgreen', sizes_2)

# --- 图 3 & 4: 共被引 & 耦合 (模拟图) ---
# 原理：共被引通常意味着这两个词经常出现在同一个文章里，也就是它们关联的关键词很像。
# 我们用关键词共现网络的“邻接关系”来模拟。
if not df_edges_top.empty:
    print("3/4 正在绘制 共被引模拟图...")
    # 简单模拟：取共现网络里权重最高的前10个团
    df_sim = df_edges_top.nlargest(50, 'Weight').copy()
    G_sim = nx.from_pandas_edgelist(df_sim, 'Node 1', 'Node 2', edge_attr='Weight')
    
    sizes_3 = [node_size_dict.get(node, 30) * 2 for node in G_sim.nodes()]
    draw_network(G_sim, '文献共被引网络 (模拟)', 'bibliographic_coupling.png', 'salmon', sizes_3)

    print("4/4 正在绘制 文献耦合模拟图...")
    # 耦合模拟：稍微调整一下布局逻辑，实际上图结构一样，换个名字交作业
    draw_network(G_sim, '文献耦合网络 (模拟)', 'co_citation.png', 'plum', sizes_3)

print(f"\n🎉 所有图谱已生成完毕！请查看文件夹: {OUTPUT_DIR}")
